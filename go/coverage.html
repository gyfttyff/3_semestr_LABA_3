type Array[T any] struct {
        data     []T
        size     int
        capacity int
}

func NewArray[T any]() *Array[T] {
        return &Array[T]{}
}

func (a *Array[T]) Init(capacity int) {
        a.data = make([]T, 0, capacity)
        a.size = 0
        a.capacity = capacity
}

func (a *Array[T]) Push(value T) {
        if a.size == a.capacity {
                a.resize()
        }
        a.data = append(a.data, value)
        a.size++
}

func (a *Array[T]) PushIdx(index int, value T) {
        if index < 0 || index > a.size {
                panic("Index out of bounds")
        }
        if a.size == a.capacity {
                a.resize()
        }
        a.data = append(a.data, *new(T))
        copy(a.data[index+1:], a.data[index:])
        a.data[index] = value
        a.size++
}

func (a *Array[T]) DelInd(index int) {
        if index < 0 || index >= a.size {
                panic("Index out of bounds")
        }
        copy(a.data[index:], a.data[index+1:])
        a.data[a.size-1] = *new(T)
        a.size--
}

func (a *Array[T]) Get(index int) T {
        if index < 0 || index >= a.size {
                panic("Index out of bounds")
        }
        return a.data[index]
}

func (a *Array[T]) Len() int {
        return a.size
}

func (a *Array[T]) Print() {
        fmt.Print("Array [", a.size, "]: ")
        for i := 0; i < a.size; i++ {
                fmt.Print(a.data[i])
                if i < a.size-1 {
                        fmt.Print(" ")
                }
        }
        fmt.Println()
}

func (a *Array[T]) Clear() {
        a.data = nil
        a.size = 0
        a.capacity = 0
}

func (a *Array[T]) resize() {
        newCapacity := a.capacity * 2
        if newCapacity == 0 {
                newCapacity = 1
        }
        newData := make([]T, a.size, newCapacity)
        copy(newData, a.data)
        a.data = newData
        a.capacity = newCapacity
}

func (a *Array[T]) SaveText(filename string) error {
        file, err := os.Create(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        fmt.Fprintln(file, a.size)
        for i := 0; i < a.size; i++ {
                fmt.Fprintln(file, a.data[i])
        }
        return nil
}

type TNode[T comparable] struct {
        data  T
        left  *TNode[T]
        right *TNode[T]
}

type CompleteBinaryTree[T comparable] struct {
        root *TNode[T]
}

func NewCompleteBinaryTree[T comparable]() *CompleteBinaryTree[T] {
        return &CompleteBinaryTree[T]{}
}

func (t *CompleteBinaryTree[T]) Insert(value T) {
        node := &TNode[T]{data: value}
        if t.root == nil {
                t.root = node
                return
        }
        queue := []*TNode[T]{t.root}
        for len(queue) > 0 {
                curr := queue[0]
                queue = queue[1:]
                if curr.left == nil {
                        curr.left = node
                        return
                }
                queue = append(queue, curr.left)
                if curr.right == nil {
                        curr.right = node
                        return
                }
                queue = append(queue, curr.right)
        }
}

func (t *CompleteBinaryTree[T]) Search(value T) bool {
        if t.root == nil {
                return false
        }
        queue := []*TNode[T]{t.root}
        for len(queue) > 0 {
                curr := queue[0]
                queue = queue[1:]
                if curr.data == value {
                        return true
                }
                if curr.left != nil {
                        queue = append(queue, curr.left)
                }
                if curr.right != nil {
                        queue = append(queue, curr.right)
                }
        }
        return false
}

func (t *CompleteBinaryTree[T]) IsComplete() bool {
        if t.root == nil {
                return true
        }
        queue := []*TNode[T]{t.root}
        foundNull := false
        for len(queue) > 0 {
                curr := queue[0]
                queue = queue[1:]
                if curr == nil {
                        foundNull = true
                } else {
                        if foundNull {
                                return false
                        }
                        queue = append(queue, curr.left)
                        queue = append(queue, curr.right)
                }
        }
        return true
}

func (t *CompleteBinaryTree[T]) Print() {
        fmt.Print("CBTree: ")

                return
        }
        queue := []*TNode[T]{t.root}

type DNode[T comparable] struct {
        data T
        prev *DNode[T]
        next *DNode[T]
}

type DoublyList[T comparable] struct {
        head *DNode[T]
        tail *DNode[T]
}

func NewDoublyList[T comparable]() *DoublyList[T] {
        return &DoublyList[T]{}
}

func (d *DoublyList[T]) Push(value T) {
        node := &DNode[T]{data: value}
        if d.tail == nil {
                d.head = node
                d.tail = node
        } else {
                node.prev = d.tail
                d.tail.next = node
                d.tail = node
        }
}

func (d *DoublyList[T]) PushHead(value T) {
        node := &DNode[T]{data: value, next: d.head}
        if d.head != nil {
                d.head.prev = node
        } else {
                d.tail = node
        }
        d.head = node
}

func (d *DoublyList[T]) Del(value T) {
        curr := d.head
        for curr != nil {
                if curr.data == value {
                        if curr.prev != nil {
                                curr.prev.next = curr.next
                        } else {
                                d.head = curr.next
                        }
                        if curr.next != nil {
                                curr.next.prev = curr.prev
                        } else {
                                d.tail = curr.prev
                        }
                        return
                }
                curr = curr.next
        }
}

func (d *DoublyList[T]) Get(value T) T {
        curr := d.head
        for curr != nil {
                if curr.data == value {
                        return curr.data
                }
                curr = curr.next
        }
        panic("Value not found")
}

func (d *DoublyList[T]) Print() {
        fmt.Print("DList: ")
        curr := d.head
        for curr != nil {
                fmt.Print(curr.data)
                if curr.next != nil {
                        fmt.Print(" <-> ")
                }
                curr = curr.next
        }
        fmt.Println(" <-> nullptr")
}

func (d *DoublyList[T]) SaveText(filename string) error {
        file, err := os.Create(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        curr := d.head
        for curr != nil {
                fmt.Fprintln(file, curr.data)
                curr = curr.next
        }
        return nil
}

func (d *DoublyList[T]) LoadText(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()
        d.head = nil
        d.tail = nil
        var val T
        for {
                if _, err := fmt.Fscan(file, &val); err != nil {
                        break
                }
                d.Push(val)
        }
        return true
}

        return d.SaveText(filename)

        next *Node[T]

}

func NewForwardList[T comparable]() *ForwardList[T] {
        return &ForwardList[T]{}
}

func (f *ForwardList[T]) PushHead(value T) {
        node := &Node[T]{data: value, next: f.head}
        f.head = node
}

func (f *ForwardList[T]) Push(value T) {
        if f.head == nil {
                f.PushHead(value)
                return
        }
        curr := f.head
        for curr.next != nil {
                curr = curr.next
        }
        curr.next = &Node[T]{data: value}
}

func (f *ForwardList[T]) Del(value T) {
        if f.head == nil {
                return
        }
        if f.head.data == value {
                f.head = f.head.next
                return
        }
        curr := f.head
        for curr.next != nil {
                if curr.next.data == value {
                        curr.next = curr.next.next
                        return
                }
                curr = curr.next
        }
}

func (f *ForwardList[T]) Get(value T) T {
        curr := f.head
        for curr != nil {
                if curr.data == value {

type HashTable struct {
        table    []string
        state    []int
        capacity int
}

func NewHashTable() *HashTable {
        return &HashTable{}
}

func (h *HashTable) Init(capacity int) {
        h.table = make([]string, capacity)
        h.state = make([]int, capacity)
        h.capacity = capacity
}

func (h *HashTable) hash(key string) int {
        hashVal := 0
        for _, c := range key {
                hashVal = hashVal*31 + int(c)
        }
        return (hashVal%h.capacity + h.capacity) % h.capacity
}

func (h *HashTable) Add(key string) {
        idx := h.hash(key)
        for h.state[idx] == 1 {
                if h.table[idx] == key {
                        return
                }
                idx = (idx + 1) % h.capacity
        }
        h.table[idx] = key
        h.state[idx] = 1
}

func (h *HashTable) Contains(key string) bool {
        idx := h.hash(key)
        startIdx := idx
        for h.state[idx] != 0 {
                if h.state[idx] == 1 && h.table[idx] == key {
                        return true
                }
                idx = (idx + 1) % h.capacity
                if idx == startIdx {
                        break
                }
        }
        return false
}

func (h *HashTable) Remove(key string) {
        idx := h.hash(key)
        startIdx := idx
        for h.state[idx] != 0 {
                if h.state[idx] == 1 && h.table[idx] == key {
                        h.state[idx] = 2
                        return
                }
                idx = (idx + 1) % h.capacity
                if idx == startIdx {
                        break
                }
        }
}

func (h *HashTable) Print() {
        fmt.Print("HashTable: ")
        for i := 0; i < h.capacity; i++ {
                if h.state[i] == 1 {
                        fmt.Printf("[%d:%s] ", i, h.table[i])
                }
        }
        fmt.Println()
}

func (h *HashTable) SaveText(filename string) error {
        file, err := os.Create(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        fmt.Fprintln(file, h.capacity)
        for i := 0; i < h.capacity; i++ {
                fmt.Fprintln(file, h.state[i])
                if h.state[i] == 1 {
                        fmt.Fprintln(file, h.table[i])
                }
        }
        return nil
}

func (h *HashTable) LoadText(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()
        var cap int
        if _, err := fmt.Fscan(file, &cap); err != nil {

        }

        }

type Queue[T any] struct {
        data     []T
        head     int
        tail     int
        capacity int
}

func NewQueue[T any]() *Queue[T] {
        return &Queue[T]{}
}

func (q *Queue[T]) Init(capacity int) {
        q.data = make([]T, capacity)
        q.head = 0
        q.tail = 0
        q.capacity = capacity
}

func (q *Queue[T]) Push(value T) {
        if (q.tail+1)%q.capacity == q.head {
                q.resize()
        }
        q.data[q.tail] = value
        q.tail = (q.tail + 1) % q.capacity
}

func (q *Queue[T]) Pop() T {
        if q.head == q.tail {
                panic("Queue is empty")
        }
        val := q.data[q.head]
        q.head = (q.head + 1) % q.capacity
        return val
}

func (q *Queue[T]) Peek() T {
        if q.head == q.tail {
                panic("Queue is empty")
        }
        return q.data[q.head]
}

func (q *Queue[T]) Len() int {
        if q.tail >= q.head {
                return q.tail - q.head
        }
        return q.capacity - q.head + q.tail
}

func (q *Queue[T]) Print() {
        fmt.Print("Queue (front to back): ")
        i := q.head
        first := true
        for i != q.tail {
                if !first {
                        fmt.Print(" ")
                }
                fmt.Print(q.data[i])
                i = (i + 1) % q.capacity
                first = false
        }
        fmt.Println()
}

func (q *Queue[T]) resize() {
        newCapacity := q.capacity * 2
        newData := make([]T, newCapacity)
        i := q.head
        j := 0
        for i != q.tail {
                newData[j] = q.data[i]
                i = (i + 1) % q.capacity
                j++
        }
        q.data = newData
        q.head = 0
        q.tail = j
        q.capacity = newCapacity
}

func (q *Queue[T]) SaveText(filename string) error {
        file, err := os.Create(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        fmt.Fprintln(file, q.Len())
        i := q.head
        for i != q.tail {
                fmt.Fprintln(file, q.data[i])
                i = (i + 1) % q.capacity
        }
        return nil
}

func (q *Queue[T]) LoadText(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()
        var size int
        if _, err := fmt.Fscan(file, &size); err != nil {
                return false
        }
        if size < 0 || size > 1000000 {
                return false
        }
        q.Init(size + 1)
        for i := 0; i < size; i++ {
                var val T
                if _, err := fmt.Fscan(file, &val); err != nil {
                        return false
                }
                q.Push(val)
        }
        return true
}

func (q *Queue[T]) SaveBin(filename string) error {
        return q.SaveText(filename)
}

        return q.LoadText(filename)

type Stack[T any] struct {
        data     []T
        top      int
        capacity int
}

func NewStack[T any]() *Stack[T] {
        return &Stack[T]{}
}

func (s *Stack[T]) Init(capacity int) {
        s.data = make([]T, capacity)
        s.top = -1
        s.capacity = capacity
}

func (s *Stack[T]) Push(value T) {
        if s.top == s.capacity-1 {
                s.resize()
        }
        s.top++
        s.data[s.top] = value
}

func (s *Stack[T]) Pop() T {
        if s.top == -1 {
                panic("Stack is empty")
        }
        val := s.data[s.top]
        s.top--
        return val
}

func (s *Stack[T]) Peek() T {
        if s.top == -1 {
                panic("Stack is empty")
        }
        return s.data[s.top]
}

func (s *Stack[T]) Len() int {
        return s.top + 1
}

func (s *Stack[T]) Print() {
        fmt.Print("Stack (top to bottom): ")
        for i := s.top; i >= 0; i-- {
                fmt.Print(s.data[i])
                if i > 0 {
                        fmt.Print(" ")
                }
        }
        fmt.Println()
}

func (s *Stack[T]) resize() {
        newCapacity := s.capacity * 2
        newData := make([]T, newCapacity)
        copy(newData, s.data)
        s.data = newData
        s.capacity = newCapacity
}

func (s *Stack[T]) SaveText(filename string) error {
        file, err := os.Create(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        fmt.Fprintln(file, s.top+1)
        for i := 0; i <= s.top; i++ {
                fmt.Fprintln(file, s.data[i])
        }
        return nil
}

func (s *Stack[T]) LoadText(filename string) bool {
        file, err := os.Open(filename)
        if err != nil {
                return false
        }
        defer file.Close()
        var size int
        if _, err := fmt.Fscan(file, &size); err != nil {
                return false
        }
        if size < 0 || size > 1000000 {
                return false
        }
        s.Init(size)
        for i := 0; i < size; i++ {
                var val T
                if _, err := fmt.Fscan(file, &val); err != nil {
                        return false
                }
                s.Push(val)
        }
        return true
}

func (s *Stack[T]) SaveBin(filename string) error {
        return s.SaveText(filename)
}

func (s *Stack[T]) LoadBin(filename string) bool {
        return s.LoadText(filename)
}
